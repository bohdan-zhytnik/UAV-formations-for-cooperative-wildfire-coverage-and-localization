cmake_minimum_required(VERSION 3.16)
project(fire_observation_demo)
find_package(ament_cmake REQUIRED)
install(DIRECTORY models worlds config launch ROMFS DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/thermal_preview.py scripts/fire_mapper.py scripts/goal_bridge.py
  scripts/fleet_router.py scripts/fleet_vehicle.py scripts/fleetctl.py
  scripts/fleet_fire_observer.py scripts/shared_fire_map.py
  scripts/mission_coordinator.py scripts/mission_cli.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/fire_geometry.py scripts/multi_fire.py scripts/mission_logic.py scripts/survey_logic.py DESTINATION lib/${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(image_transport REQUIRED)
find_package(message_filters REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(fire_rviz src/fire_rviz.cpp)
target_compile_features(fire_rviz PRIVATE cxx_std_17)
ament_target_dependencies(fire_rviz rclcpp rviz_common std_msgs)
target_link_libraries(fire_rviz Qt5::Widgets)
install(TARGETS fire_rviz DESTINATION lib/${PROJECT_NAME})
add_library(fleet_goal_tool SHARED src/fleet_goal_tool.cpp src/fleet_image_display.cpp)
target_compile_features(fleet_goal_tool PRIVATE cxx_std_17)
ament_target_dependencies(fleet_goal_tool rclcpp rviz_common rviz_rendering rviz_default_plugins std_msgs pluginlib image_transport message_filters)
target_link_libraries(fleet_goal_tool Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common fleet_tools.xml)
install(TARGETS fleet_goal_tool LIBRARY DESTINATION lib)
add_executable(fleet_rviz src/fleet_rviz.cpp)
target_compile_features(fleet_rviz PRIVATE cxx_std_17)
ament_target_dependencies(fleet_rviz rclcpp rviz_common std_msgs)
target_link_libraries(fleet_rviz Qt5::Widgets fleet_goal_tool)
install(TARGETS fleet_rviz DESTINATION lib/${PROJECT_NAME})
add_executable(survey_tool_check src/survey_tool_check.cpp)
target_compile_features(survey_tool_check PRIVATE cxx_std_17)
ament_target_dependencies(survey_tool_check rclcpp rviz_common std_msgs)
target_link_libraries(survey_tool_check Qt5::Widgets fleet_goal_tool)
install(TARGETS survey_tool_check DESTINATION lib/${PROJECT_NAME})
ament_package()
